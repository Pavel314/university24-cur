: star_table  #0x084A, table of stars (x,y) 16 total
  0x0A 0x11  0x5B 0x22  0x5B 0x29  0x75 0x38 
	0x54 0x11  0x0C 0x09  0x34 0x13  0x78 0x34 
	0x26 0x04  0x06 0x1A  0x0C 0x22  0x20 0x0A 
	0x00 0x16  0x06 0x05  0x3F 0x07  0x00 0x1E


:alias star_y v1
:alias mask v3
:alias counter v4
:const stars_count 16

: draw_star_i
  i := star_table
  #v0 <<= v5
  v0 += v0
	i += v0
  load star_y
  i := 0x0C39
  sprite v0 star_y 0x01
  return


: init_stars
  i := 0x0C56
  v0 := 0x00
  save v0
  v2 := v0
  counter := 0
  : rp1
  draw_star_i
  v2 += 0x01
  v0 := v2
  counter += 1
  if counter != stars_count then jump rp1
	return

: wait
  v0 := 0x01
  delay := v0
  : rp2
  v0 := delay
  if v0 != 0x00 then jump rp2
  return

: draw_stars
  i := 0x0C56
  load v0
  v2 := v0
  v4 := v0
  draw_star_i
  i := star_table
  #v2 <<= v5
  v2 += v2
	i += v2
  v0 := random 0b01111111
  star_y := random 0b00111111
  save star_y
  i := 0x0C39
  sprite v0 star_y 0x01
  mask := 0x0F
  v4 += 0x01
  v4 &= mask
  v0 := v4
  i := 0x0C56
  save v0
  return  


: main
   clear
	 
   i := 0x0C39
	 v0 := 0x80
	 save v0
	 
   init_stars
   loop
   wait
   draw_stars
   v0 := 0x0A
   again
