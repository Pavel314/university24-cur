: main
    hires

    v0 := 0     # x
    v1 := 8     # y
    v2 := 0     # digit counter (0–9)

    loop
		 v3 := v2
		 v3 += 0
     i := bighex v3   # Fx30 — указатель на hi-res цифру (0–9)
     sprite v0 v1 10  # 10 строк = hi-res glyph

     v0 += 11         # сдвиг по X (10px glyph + отступ)
     v2 += 1
     if v2 < 11 then again

: halt
    jump halt
